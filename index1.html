        // PROGRESS CALCULATIONS

        function updateProgress() {

            const athlete = queue[activeIndex];

            if(!athlete) return;

 

            const totals = selectedTray.reduce((acc, curr) => ({

                kcal: acc.kcal + curr.kcal,

                glucid: acc.glucid + curr.glucid,

                protein: acc.protein + curr.protein,

                water: acc.water + curr.water

            }), { kcal: 0, glucid: 0, protein: 0, water: 0 });

 

            // Kcal

            document.getElementById('kcalVal').innerText = `${totals.kcal} / ${athlete.target.kcal}`;

            document.getElementById('kcalBar').style.width = `${Math.min(100, (totals.kcal / athlete.target.kcal) * 100)}%`;

 

            // Glucid

            document.getElementById('glucidVal').innerText = `${totals.glucid.toFixed(1)}g / ${athlete.target.glucid}g`;

            document.getElementById('glucidBar').style.width = `${Math.min(100, (totals.glucid / athlete.target.glucid) * 100)}%`;

 

            // Protein

            document.getElementById('proteinVal').innerText = `${totals.protein.toFixed(1)}g / ${athlete.target.protein}g`;

            document.getElementById('proteinBar').style.width = `${Math.min(100, (totals.protein / athlete.target.protein) * 100)}%`;

 

            // Water

            document.getElementById('waterVal').innerText = `${totals.water}ml / ${athlete.target.water}ml`;

            document.getElementById('waterBar').style.width = `${Math.min(100, (totals.water / athlete.target.water) * 100)}%`;

        }

                    timerInterval = null;

                    document.getElementById('startBtnText').innerText = "Bắt đầu (60s)";

                }

            }, 1000);

        function calculateFoodConversion() {

            const selectedId = document.getElementById('convertFoodSelect').value;

            const weight = parseFloat(document.getElementById('convertWeight').value) || 0;

            const item = rawIngredients.find(r => r.id === selectedId) || rawIngredients[0];

 

            const ratio = weight / 100;

            const resKcal = (item.kcal * ratio).toFixed(1);

            const resGlucid = (item.glucid * ratio).toFixed(1);

            const resProtein = (item.protein * ratio).toFixed(1);

            const resWater = (item.water * ratio).toFixed(1);

 

            document.getElementById('conversionResult').innerHTML = `

                <div class="font-bold text-slate-800 text-xs">${weight}g/ml ${item.name} chứa:</div>

                <div class="grid grid-cols-2 gap-1 text-slate-600">

                    <div>• Calo: <strong class="text-amber-600">${resKcal} kcal</strong></div>

                    <div>• Glucid: <strong>${resGlucid} g</strong></div>

                    <div>• Protein: <strong>${resProtein} g</strong></div>

                    <div>• Lượng nước: <strong>${resWater} ml</strong></div>

                </div>

            `;

        }

        function calculateBMI() {

            const h = parseFloat(document.getElementById('bmiHeight').value) / 100;

            const w = parseFloat(document.getElementById('bmiWeight').value);

            const ageGroup = document.getElementById('bmiAgeGroup').value;

 

            if(!h || !w || h <= 0 || w <= 0) return;

 

            const bmi = (w / (h * h)).toFixed(1);

            let state = "";

            let color = "";

            let dietAdvice = "";

            let exerciseAdvice = "";

 

            if(bmi < 18.5) {

                state = "Thiếu cân";

                color = "text-amber-600";

                dietAdvice = "Bổ sung các món ăn giàu đạm & năng lượng lành mạnh như Phở gà, Trứng gà, Sữa đậu nành, Xôi xéo.";

                exerciseAdvice = "Tập thể thao vừa sức 30-45 phút/ngày (Bơi lội, Nhảy dây, Đạp xe) để kích thích thèm ăn và tăng mật độ xương.";

            } else if(bmi >= 18.5 && bmi < 23) {

                state = "Bình thường (Khỏe mạnh)";

                color = "text-emerald-600";

                dietAdvice = "Duy trì chế độ ăn cân đối 55% Glucid, 20% Protein, bổ sung Chuối ngự & Nước dừa tươi hàng ngày.";

                exerciseAdvice = "Duy trì ít nhất 60 phút vận động thể thao mỗi ngày (Bóng đá, Cầu lông, Bơi lội) để phát triển thể chất toàn diện.";

            } else if(bmi >= 23 && bmi < 25) {

                state = "Tiền béo phì";

                color = "text-orange-600";

                dietAdvice = "Giảm bớt đồ ngọt và tinh bột đậm đặc (chè đỗ đen, xôi), tăng cường khoai lang luộc, rau xanh và nước lọc.";

            } else {

                state = "Béo phì";

            }

                    <span class="font-bold text-slate-700">Chỉ số BMI: <strong class="text-base">${bmi}</strong></span>
